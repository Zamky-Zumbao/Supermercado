// Script F12 - Extrae nombre, precio e imagen de cada producto
(async function() {
    console.log('🔍 Extrayendo productos con imágenes...');
    
    const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms));
    const productos = [];
    let productosAnteriores = 0;
    let sinCambios = 0;
    let scrolls = 0;
    
    // === 1. SCROLL AUTOMÁTICO ===
    while (scrolls < 30 && sinCambios < 4) {
        window.scrollTo(0, document.documentElement.scrollHeight);
        await wait(3000);
        
        const imagenes = document.querySelectorAll('img[alt]');
        let count = 0;
        imagenes.forEach(img => {
            const alt = img.getAttribute('alt');
            if (alt && alt.length > 10 && !alt.toLowerCase().includes('logo')) {
                count++;
            }
        });
        console.log(`   Scroll ${scrolls + 1}: ${count} imágenes de productos`);
        
        if (count === productosAnteriores) {
            sinCambios++;
        } else {
            sinCambios = 0;
            productosAnteriores = count;
        }
        scrolls++;
    }
    
    // === 2. EXTRAER DATOS ===
    const imagenes = document.querySelectorAll('img[alt]');
    console.log(`\n📦 Procesando ${imagenes.length} imágenes...`);
    
    imagenes.forEach(img => {
        const alt = img.getAttribute('alt');
        const src = img.getAttribute('src');
        
        if (!alt || alt.length < 10 || alt.toLowerCase().includes('logo')) return;
        if (!src) return;
        
        // Buscar precio
        let precio = null;
        let padre = img.parentElement;
        for (let i = 0; i < 8 && padre; i++) {
            const match = padre.innerText.match(/\$\s*([\d,.]+)/);
            if (match) {
                precio = parseInt(match[1].replace(/\./g, '').replace(/,/g, ''));
                break;
            }
            padre = padre.parentElement;
        }
        
        // Limpiar nombre
        let nombre = alt.trim()
            .replace(/\s+contenido\s+neto\s+[\d.]+\s*[A-Za-z]+/i, '')
            .replace(/\s+Contenido\s+drenado\s+[\d.]+\s*[A-Za-z]+/i, '')
            .trim();
        
        if (nombre.length > 5 && precio && precio > 100 && precio < 1000000) {
            // Verificar si ya existe (evitar duplicados)
            const existe = productos.find(p => p.nombre === nombre);
            if (!existe) {
                productos.push({
                    nombre: nombre,
                    precio: precio,
                    imagen: src
                });
            } else if (precio < existe.precio) {
                // Actualizar con precio más bajo
                existe.precio = precio;
                existe.imagen = src;
            }
        }
    });
    
    // Ordenar por nombre
    productos.sort((a, b) => a.nombre.localeCompare(b.nombre));
    
    console.log(`\n✅ PRODUCTOS ENCONTRADOS: ${productos.length}`);
    console.table(productos.slice(0, 10));
    
    // === 3. DESCARGAR ARCHIVO ===
    const json = JSON.stringify(productos, null, 2);
    const blob = new Blob([json], {type: 'application/json'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'productos_con_imagenes.json';
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
    console.log('\n📥 Archivo descargado: productos_con_imagenes.json');
    
    // === 4. GUARDAR EN VARIABLE ===
    window.productosConImagenes = productos;
    console.log('📋 También disponible en: window.productosConImagenes');
    
    return productos;
})();
